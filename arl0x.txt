# ---------- text helpers ----------
trunc(){ local s=$1 m=$2
  [ "${#s}" -gt "$m" ] && s="${s:0:$((m-1))}$ELL"
  printf '%s' "$s"; }
mmss(){ printf '%02d:%02d' $(( ${1:-0}/60 )) $(( ${1:-0}%60 )); }
hrule(){ local n=$1 s; printf -v s '%*s' "$n" ''; printf '%s' "${s// /$GH}"; }

# ---------- screen engine: compose -> single-write draw ----------
scr(){ SCR+=("$1"); }
scr_cursor(){ CURSOR="$1;$2"; }
draw(){
  if [ "$FIRSTDRAW" -eq 1 ]; then printf '\e[2J'; FIRSTDRAW=0; fi
  printf '\e[H'
  local i=0 n=${#SCR[@]}
  while [ "$i" -lt "$n" ]; do
    if [ $((i+1)) -lt "$n" ]; then printf '%s\e[K\n' "${SCR[$i]}"
    else printf '%s\e[K' "${SCR[$i]}"; fi
    i=$((i+1))
  done
  printf '\e[J'
  if [ -n "$CURSOR" ]; then printf '\e[%sH\e[?25h' "$CURSOR"
  else printf '\e[?25l'; fi
}

# ---------- frame builders ----------
frame_open(){ local t r
  t=$(trunc "$1" $((COLS-14)))
  r=$((COLS-7-${#t})); [ "$r" -lt 1 ] && r=1
  scr "$DG$GTL$GH$RST $AQUA$DIA$RST${2:-$B$TEAL} $t$RST $DG$(hrule $r)$GTR$RST"; }
frame_blank(){ scr "$DG$GV$RST$(printf '%*s' $((COLS-2)) '')$DG$GV$RST"; }
frame_div(){ scr "$DG$GDL$(hrule $((COLS-2)))$GDR$RST"; }
frame_close(){ scr "$DG$GBL$(hrule $((COLS-2)))$GBR$RST"; }
line_in(){ local pad=$((COLS-4-$2)); [ "$pad" -lt 1 ] && pad=1
  scr "$DG$GV$RST  $1$RST$(printf '%*s' "$pad" '')$DG$GV$RST"; }
frow(){ local k v
  k=$(printf '%-8s' "$1"); v=$(trunc "$2" $((COLS-20)))
  line_in "$GRAY$k$RST ${3:-$INK}$v$RST" "$((9+${#v}))"; }
ftext(){ local t ind pad
  ind=${3:-2}
  t=$(trunc "$1" $((COLS-6-ind)))
  pad=$((COLS-2-ind-${#t})); [ "$pad" -lt 1 ] && pad=1
  scr "$DG$GV$RST$(printf '%*s' "$ind" '')${2:-$GRAY}$t$RST$(printf '%*s' "$pad" '')$DG$GV$RST"; }
cell(){ local lbl; lbl=$(printf '%-7s' "$2")
  printf '%s%s%s %s%s%s' "$B$AQUA" "$1" "$RST" "$GRAY" "$lbl" "$RST"; }
leg(){ local c1 c2 c3 P
  c1=$(cell "$1" "$2"); c2=$(cell "$3" "$4"); c3=''
  P="$1 $(printf '%-7s' "$2")  $3 $(printf '%-7s' "$4")"
  if [ -n "${5:-}" ]; then c3=$(cell "$5" "$6"); P="$P  $5 $(printf '%-7s' "$6")"; fi
  line_in "$c1  $c2${c3:+  $c3}$RST" "${#P}"; }
hdr(){ scr ''; scr "  $AQUA$DIA$RST ${B}ar0x sherlock${RST}${DG} v$VERSION $DOT $1${RST}"; scr ''; }

# mixed-color row builder: seg <text> <color> ... then emit_row
PB=''; CB=''
seg(){ PB="$PB$1"; CB="$CB$2$1$RST"; }
emit_row(){

  else line_in "$CB" "${#PB}"; fi
  PB=''; CB=''; }

# ---------- THE GRID · live site status map ----------
# no associative arrays here — old bash + set -u treats a declared-but-
# empty assoc array as unbound and crashes. two parallel indexed arrays
# and in-process string matching instead.
grid_load(){
  local res nm st entry i n
  res=$(cat "$RESF" 2>/dev/null)
  GORDER=(); GST=()
  GCOLS=$(( (COLS-8)/13 )); [ "$GCOLS" -lt 1 ] && GCOLS=1
  local -a TN=() TS=()
  for entry in ${SITES[@]+"${SITES[@]}"}; do
    nm=${entry%%|*}
    st=P
    case "$res" in
      *"FOUND|$nm|"*) st=FOUND ;;
      *"NO|$nm|"*)    st=NO ;;
      *"ERR|$nm|"*)   st=ERR ;;
    esac
    TN+=("$nm"); TS+=("$st")
  done
  # display order: found first, then in-flight, then miss, then blocked
  n=${#TN[@]}
  for ((i=0;i<n;i++)); do [ "${TS[$i]}" = FOUND ] && { GORDER+=("${TN[$i]}"); GST+=(FOUND); }; done
  for ((i=0;i<n;i++)); do [ "${TS[$i]}" = P ]     && { GORDER+=("${TN[$i]}"); GST+=(P); }; done
  for ((i=0;i<n;i++)); do [ "${TS[$i]}" = NO ]    && { GORDER+=("${TN[$i]}"); GST+=(NO); }; done
  for ((i=0;i<n;i++)); do [ "${TS[$i]}" = ERR ]   && { GORDER+=("${TN[$i]}"); GST+=(ERR); }; done
  return 0
}

grid_render(){ # $1 rows visible  $2 pulse frame  $3 start index (for scrolling)
  local rows=$1 fr=$2 start=$3 r c idx name g col line cnt
  for (( r=0; r<rows; r++ )); do
    line=''; cnt=0
    for (( c=0; c<GCOLS; c++ )); do
      idx=$(( start + r*GCOLS + c ))
      [ "$idx" -ge "${#GORDER[@]}" ] && break
      name=${GORDER[$idx]}
      case "${GST[$idx]}" in
        FOUND) g=$GF
               if [ $(( fr % 2 )) -eq 0 ]; then col="$B$AQUA"; else col="$TEAL"; fi ;;
        NO)    g=$GM; col="$DG" ;;
        ERR)   g=$XGL; col="$RED" ;;
        *)     g=$GP; col="$DG" ;;
      esac
      line="$line$col$g $(printf '%-10s' "$(trunc "$name" 10)")$RST "
      cnt=$((cnt+1))
    done
    [ "$cnt" -gt 0 ] && line_in "$line" "$(( cnt*13 ))"
  done
  return 0
}

# full-screen scrollable sweep grid (g on the results screen)
grid_view(){
  local OFF=0 rows totalg maxoff
  while :; do
    grid_load
    totalg=${#GORDER[@]}
    rows=$(( LINES - 14 ))
    [ "$rows" -lt 2 ] && rows=2
    [ "$rows" -gt 16 ] && rows=16
    maxoff=$(( (totalg + GCOLS - 1) / GCOLS - rows ))
    [ "$maxoff" -lt 0 ] && maxoff=0
    [ "$OFF" -gt "$maxoff" ] && OFF=$maxoff
    [ "$OFF" -lt 0 ] && OFF=0
    SCR=(); CURSOR=''
    hdr "the sweep $DOT $UNAME"
    frame_open 'THE GRID'
    frame_blank
    frow 'target' "$UNAME" "$B$AQUA"
    frame_blank
    grid_render "$rows" 0 $(( OFF*GCOLS ))
    frame_blank
    ftext "$GF found  $DOT  $XGL blocked  $DOT  $GM miss  $DOT  $GP never resolved" "$DG"
    frame_close
    scr ''
    scr "  $D↑↓ scroll $DOT any other key goes back$RST"
    draw
    read_key || return 0
    case "$KEY" in
      UP|k|K)   OFF=$((OFF-1)); [ "$OFF" -lt 0 ] && OFF=0 ;;
      DOWN|j|J) OFF=$((OFF+1)); [ "$OFF" -gt "$maxoff" ] && OFF=$maxoff ;;
      *)        return 0 ;;
    esac
  done
}

# ---------- key reader ----------
read_key(){
  local k k2 k3
  KEY=''
  IFS= read -rsn1 k || return 1
  if [ "$k" = $'\e' ]; then
    IFS= read -rsn1 -t 0.05 k2 2>/dev/null || k2=''
    if [ "$k2" = '[' ]; then
      IFS= read -rsn1 -t 0.05 k3 2>/dev/null || k3=''
      case "$k3" in A) KEY=UP;; B) KEY=DOWN;; *) KEY=ESC;; esac
    else KEY=ESC; fi
  else KEY=$k; fi
  return 0
}

  frame_blank

# ---------- step 1 · the username ----------
ask_username(){
  local input pv
  while :; do
    SCR=(); CURSOR=''
    banner_lines
    frame_open 'USERNAME TO HUNT'
    frame_blank
    if [ -n "$LAST_USER" ]; then
      ftext "last hunt: $LAST_USER $DASH enter repeats it" "$DG"
      frame_blank
    fi
    pv="$GV  $TRI "
    scr_cursor "$(( ${#SCR[@]} + 1 ))" "$(( ${#pv} + 1 ))"
    scr "$DG$GV$RST  $AQUA$TRI$RST "
    frame_blank
    frame_close
    scr ''
    scr "  $D$(trunc 'letters, numbers, dots, _ and - · 2 to 32 chars' $((COLS-4)))$RST"
    draw
    IFS= read -r input || { printf '\e[?25h\n'; exit 0; }
    printf '\e[?25l'
    [ -z "$input" ] && [ -n "$LAST_USER" ] && input="$LAST_USER"
    if [[ "$input" =~ ^[A-Za-z0-9._-]{2,32}$ ]]; then
      UNAME=$input
      return 0
    fi
    SCR=(); CURSOR=''
    banner_lines
    frame_open 'USERNAME TO HUNT'
    frame_blank
    ftext "$XGL  usernames are 2-32 chars: a-z 0-9 . _ -" "$RED"
    frame_blank
    frame_close
    draw
    sleep 1.1
  done
}

# ---------- one site check (runs in background) ----------
check_one(){
  local name=$1 url=$2 mode=$3 param=$4 code body
  url=${url//'{u}'/$UNAME}
  if [ "$mode" = C ]; then
    code=$(curl -A "$UA" -s -o /dev/null -m "$TMO" -w '%{http_code}' -L "$url")
    case "$code" in
      200) printf 'FOUND|%s|%s\n' "$name" "$url" >>"$RESF" ;;
      404) printf 'NO|%s|%s\n'    "$name" "$url" >>"$RESF" ;;
      *)   printf 'ERR|%s|%s\n'   "$name" "$url" >>"$RESF" ;;
    esac
  else
    if body=$(curl -A "$UA" -s -m "$TMO" -L "$url"); then
      if printf '%s' "$body" | grep -qiF "$param"; then
        printf 'NO|%s|%s\n' "$name" "$url" >>"$RESF"
      else
        printf 'FOUND|%s|%s\n' "$name" "$url" >>"$RESF"
      fi
    else
      printf 'ERR|%s|%s\n' "$name" "$url" >>"$RESF"
    fi
  fi
}

# ---------- live scan screen ----------
draw_scan(){ # $1 spinner frame
  local fr=$1 found no err done_ pct i fill barC
  grid_load
  found=$(grep -c '^FOUND|' "$RESF" 2>/dev/null); found=${found:-0}
  no=$(grep -c '^NO|' "$RESF" 2>/dev/null);       no=${no:-0}
  err=$(grep -c '^ERR|' "$RESF" 2>/dev/null);     err=${err:-0}
  done_=$((found+no+err))
  pct=$(( done_*100/TOTAL ))
  SCR=(); CURSOR=''
  hdr "hunting $UNAME on $TOTAL sites"
  frame_open "HUNTING $DOT $UNAME"
  frame_blank
  PB=''; CB=''
  seg "${SPIN[$fr]}  " "$B$AQUA"
  seg "checked $done_/$TOTAL" "$B$INK"
  seg "   $DOT   " "$DG"
  seg "found $found" "$AQUA"
  seg "   $DOT   " "$DG"
  seg "miss $no" "$GRAY"
  seg "   $DOT   " "$DG"
  seg "blocked $err" "$DG"
  seg "   $DOT   " "$DG"
  seg "$(mmss "$SECONDS")" "$GRAY"
  emit_row
  frame_blank
  local -a LV=()
  mapfile -t LV < <(tail -n 3 "$RESF" 2>/dev/null)
  local l st nm col msg
  for l in ${LV[@]+"${LV[@]}"}; do
    st=${l%%|*}
    nm=$(printf '%s' "$l" | cut -d'|' -f2)
    case "$st" in
      FOUND) col=$AQUA; msg='found' ;;
      NO)    col=$GRAY; msg='no trace' ;;
      *)     col=$RED;  msg='blocked' ;;
    esac
    nm=$(trunc "$nm" $((COLS-18)))
    PB=''; CB=''
    seg "$ARR " "$DG"
    seg "$nm" "$col"
    seg "  $DASH  $msg" "$GRAY"
    emit_row
  done
  [ "${#LV[@]}" -eq 0 ] && ftext "the first checks are in flight$ELL" "$DG"
  frame_blank
  fill=$(( done_*BARW/TOTAL )); [ "$fill" -gt "$BARW" ] && fill=$BARW
  barC=''
  for ((i=0;i<fill;i++)); do barC="$barC$AQUA$BARF"; done
  for ((i=fill;i<BARW;i++)); do barC="$barC$DG$BARE"; done
  line_in "$barC$RST  $GRAY$pct%$RST" "$((BARW+3+${#pct}))"
  frame_blank
  local -a FN=()
  mapfile -t FN < <(grep '^FOUND|' "$RESF" 2>/dev/null | cut -d'|' -f2 | tail -n 5)
  if [ "${#FN[@]}" -gt 0 ]; then
    local joined IFS=' · '
    joined="${FN[*]}"
    joined=$(trunc "$joined" $((COLS-22)))
    PB=''; CB=''
    seg 'found so far   ' "$DG"
    seg "$joined" "$AQUA"
    emit_row
  else
    ftext "nothing confirmed yet $DASH the checks are in flight" "$DG"
  fi
  frame_blank
  if [ "$INTR" -eq 1 ]; then
    ftext "interrupted $DASH partial results are kept" "$RED"
  else
    ftext "ctrl+c stops the hunt $DASH partial results are kept" "$DG"
  fi
  frame_close

  # ---- THE GRID · every site, live status, fills the rest of the screen ----
  local avail=$(( LINES - ${#SCR[@]} - 4 ))
  if [ "$avail" -ge 2 ]; then
    scr ''
    frame_open 'THE GRID'
    grid_render "$avail" "$fr" 0
    local totalg=${#GORDER[@]} vis=$(( avail * GCOLS )) more=''
    [ "$vis" -lt "$totalg" ] && more="  $DOT  $ELL$(( totalg - vis )) below"
    ftext "$GF found  $DOT  $XGL blocked  $DOT  $GM miss  $DOT  $GP in flight$more" "$DG"
    frame_close
  fi
  draw
}

scan(){
  local entry n u m p fr=0
  SECONDS=0
  INTR=0
  : >"$RESF"
  SCANNING=1
  for entry in "${SITES[@]}"; do
    [ "$INTR" -eq 1 ] && break
    IFS='|' read -r n u m p <<<"$entry"
    while [ "$(jobs -rp | wc -l)" -ge "$MAXJ" ]; do
      draw_scan "$fr"; fr=$(( (fr+1)%${#SPIN[@]} ))
      BSTATE=$(( (BSTATE+1)%BARW ))
      sleep 0.12
      [ "$INTR" -eq 1 ] && break
    done
    [ "$INTR" -eq 1 ] && break
    check_one "$n" "$u" "$m" "$p" &
    draw_scan "$fr"; fr=$(( (fr+1)%${#SPIN[@]} ))
  done
  while [ "$(jobs -rp | wc -l)" -gt 0 ] && [ "$INTR" -eq 0 ]; do
    draw_scan "$fr"; fr=$(( (fr+1)%${#SPIN[@]} ))
    BSTATE=$(( (BSTATE+1)%BARW ))
    sleep 0.12
  done
  if [ "$INTR" -eq 1 ]; then
    jobs -p 2>/dev/null | while IFS= read -r pid; do kill "$pid" 2>/dev/null; done
    sleep 0.2
  fi
  SCANNING=0
}

# ---------- results ----------
open_url(){
  if command -v xdg-open >/dev/null 2>&1; then
    ( xdg-open "$1" >/dev/null 2>&1 ) &
  else
    show_err 'no xdg-open here — the url is shown under the list'
  fi
}

save_report(){
  local out dir="$HOME/Downloads" nf nn ne
  mkdir -p "$dir" 2>/dev/null || dir=$PWD
  out="$dir/sherlock-$UNAME-$(date +%Y%m%d-%H%M).txt"
  nf=$(grep -c '^FOUND|' "$RESF" 2>/dev/null); nf=${nf:-0}
  nn=$(grep -c '^NO|' "$RESF" 2>/dev/null);    nn=${nn:-0}
  ne=$(grep -c '^ERR|' "$RESF" 2>/dev/null);   ne=${ne:-0}
  {
    printf 'ar0x sherlock report\n'
    printf 'date      %s\n' "$(date '+%Y-%m-%d %H:%M')"
    printf 'username  %s\n' "$UNAME"
    printf 'checked   %s sites\n\n' "$TOTAL"
    printf 'FOUND (%s)\n' "$nf"
    grep '^FOUND|' "$RESF" | sort -t'|' -k2,2 | while IFS='|' read -r _ nm ur; do
      printf '  %-16s %s\n' "$nm" "$ur"
    done
    printf '\nNO TRACE (%s)\n' "$nn"
    grep '^NO|' "$RESF" | cut -d'|' -f2 | while IFS= read -r nm; do printf '  %s\n' "$nm"; done
    printf '\nBLOCKED / UNKNOWN (%s)\n' "$ne"
    grep '^ERR|' "$RESF" | cut -d'|' -f2 | while IFS= read -r nm; do printf '  %s\n' "$nm"; done
    printf '\ngenerated by ar0x sherlock v%s — github.com/ar0x1\n' "$VERSION"
  } >"$out" 2>/dev/null
  SCR=(); CURSOR=''
  hdr 'report written'
  frame_open 'REPORT SAVED'
  frame_blank
  frow 'wrote' "$(trunc "$out" $((COLS-20)))" "$AQUA"
  frow 'holds' 'found · clean · blocked'
  frame_blank
  ftext 'press any key to go back' "$DG"
  frame_close
  draw
  read_key || true
}

results_browse(){
  local -a FL_=() NL_=() EL_=()
  mapfile -t FL_ < <(grep '^FOUND|' "$RESF" 2>/dev/null | sort -t'|' -k2,2)
  mapfile -t NL_ < <(grep '^NO|' "$RESF" 2>/dev/null | cut -d'|' -f2)
  mapfile -t EL_ < <(grep '^ERR|' "$RESF" 2>/dev/null | cut -d'|' -f2)
  local nf=${#FL_[@]} nn=${#NL_[@]} ne=${#EL_[@]}
  if [ "$nf" -eq 0 ]; then
    SCR=(); CURSOR=''
    hdr 'the hunt is over'
    frame_open 'NO TRACES FOUND'
    frame_blank
    frow 'username' "$UNAME" "$B$AQUA"
    frow 'checked' "$TOTAL sites"
    frow 'found'   '0 profiles' "$AQUA"
    [ "$INTR" -eq 1 ] && frow 'note' 'interrupted — partial results' "$RED"
    frame_blank
    ftext 'either the name is genuinely free, or it hides behind' "$GRAY"
    ftext 'walls this hunt cannot climb (facebook, x, instagram)' "$GRAY"
    frame_blank
    ftext 'g the sweep grid  ·  n new hunt  ·  any other key exits' "$DG"
    frame_close
    draw
    read_key || return 0
    case "$KEY" in
      g|G) grid_view ;;
      n|N) return 1 ;;
      *)   return 0 ;;
    esac
  fi
  local SEL=0 OFF=0 PSZ=$(( LINES-17 )) i nm pfx col end selurl
  [ "$PSZ" -lt 3 ] && PSZ=3
  [ "$PSZ" -gt 12 ] && PSZ=12
  while :; do
    [ "$SEL" -ge "$nf" ] && SEL=$((nf-1))
    [ "$SEL" -lt 0 ] && SEL=0
    [ "$OFF" -gt $(( nf-PSZ )) ] && OFF=$(( nf-PSZ ))
    [ "$OFF" -lt 0 ] && OFF=0
    [ "$SEL" -lt "$OFF" ] && OFF=$SEL
    [ "$SEL" -ge $(( OFF+PSZ )) ] && OFF=$(( SEL-PSZ+1 ))
    SCR=(); CURSOR=''
    hdr 'the hunt is over'
    frame_open "FOUND $DOT ${nf} PROFILES"
    frame_blank
    frow 'username' "$UNAME" "$B$AQUA"
    frow 'found'    "$nf profiles" "$AQUA"
    frow 'miss'     "$nn clean"
    frow 'blocked'  "$ne unknown"
    [ "$INTR" -eq 1 ] && frow 'note' 'interrupted — partial results' "$RED"
    frame_blank
    frame_div
    end=$(( OFF+PSZ )); [ "$end" -gt "$nf" ] && end=$nf
    for (( i=OFF; i<end; i++ )); do
      nm=$(printf '%s' "${FL_[$i]}" | cut -d'|' -f2)
      nm=$(trunc "$nm" $((COLS-12)))
      if [ "$i" -eq "$SEL" ]; then pfx="$AQUA$TRI$RST "; col=$B$INK
      else pfx='  '; col=$GRAY; fi
      line_in "$pfx$col$nm$RST" "$((2+${#nm}))"
    done
    [ "$nf" -gt "$end" ] && ftext "$ELL $(( nf-end )) more below" "$DG"
    frame_blank
    frame_div
    selurl=$(printf '%s' "${FL_[$SEL]}" | cut -d'|' -f3)
    ftext "$ARR $(trunc "$selurl" $((COLS-10)))" "$AQUA"
    frame_blank
    leg '↑↓' 'move' 'enter' 'open' 'g' 'grid'
    leg s 'save' n 'new hunt' q 'quit'
    frame_close
    draw
    read_key || return 0
    case "$KEY" in
      UP|k|K)   [ "$SEL" -gt 0 ] && SEL=$((SEL-1)) ;;
      DOWN|j|J) [ "$SEL" -lt $((nf-1)) ] && SEL=$((SEL+1)) ;;
      ''|o|O)   open_url "$selurl" ;;
      g|G)      grid_view ;;
      s|S)      save_report ;;
      n|N)      return 1 ;;
      q|Q)      return 0 ;;
    esac
  done
}

cline(){ local p=$(( (COLS-$2)/2 )); [ "$p" -lt 0 ] && p=0
  printf '%*s%s\n' "$p" '' "$1"; }
goodbye(){

  printf '\n'
  cline "$D$GRAY a name is a trail — follow it.$RST" 29

  cline "$D in the spirit of the sherlock project$RST" 37

}

# ---------- main ----------
main(){
  local pre='' rc
  while [ $# -gt 0 ]; do
    case "$1" in
      -h|--help)    usage; exit 0 ;;
      -v|--version) printf 'ar0x sherlock %s\n' "$VERSION"; exit 0 ;;
      -*)           printf 'unknown option: %s\n\n' "$1"; usage; exit 1 ;;
      *)            pre="$1" ;;
    esac
    shift
  done

  RESF=$(mktemp /tmp/ar0xsherlock.XXXXXX 2>/dev/null) || RESF="/tmp/ar0xsherlock.$$"
  trap 'printf "\e[?25h\e[0m\n"; rm -f "$RESF"' EXIT
  on_int(){
    if [ "${SCANNING:-0}" -eq 1 ]; then
      INTR=1
      jobs -p 2>/dev/null | while IFS= read -r pid; do kill "$pid" 2>/dev/null; done
      printf '\e[?25h'
    else
      printf '\e[?25h'
      printf '\n\n%s%s interrupted%s\n' "$RED" "$XGL" "$RST"
      exit 130
    fi
  }
  trap on_int INT TERM

  need_curl
  load_last
  UNAME=''

  if [ -n "$pre" ]; then
    if [[ "$pre" =~ ^[A-Za-z0-9._-]{2,32}$ ]]; then UNAME=$pre
    else printf '%s\n' "${RED}${XGL} '$pre' is not a valid username (a-z 0-9 . _ - , 2-32 chars)${RST}"; exit 1; fi
  fi

  while :; do
    [ -z "$UNAME" ] && { ask_username || break; }
    scan
    mkdir -p "$AX_DIR" 2>/dev/null
    printf '%s\t%s\n' "$(date '+%Y-%m-%d %H:%M')" "$UNAME" >>"$AX_HISTORY" 2>/dev/null
    LAST_USER="$UNAME"
    results_browse; rc=$?
    [ "$rc" -eq 0 ] && break
    UNAME=''
  done

  goodbye
}
main "$@"
